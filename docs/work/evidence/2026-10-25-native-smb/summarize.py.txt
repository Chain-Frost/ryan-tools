from pathlib import Path
import json,statistics
ROOT=Path(__file__).absolute().parent
rows=[json.loads(p.read_text()) for p in (ROOT/'reports').glob('*/result.json')]
def stats(values):
 return {'n':len(values),'median':statistics.median(values),'mean':statistics.mean(values),'min':min(values),'max':max(values),'sample_sd':statistics.stdev(values) if len(values)>1 else None,'values':values}
def selected(stage,variant,condition):return sorted([r for r in rows if r['stage']==stage and r['variant']==variant and r['condition']==condition],key=lambda r:r['trial'])
result={'campaign_completed':(ROOT/'completed.json').exists(),'full_cross_revision_parity':(ROOT/'cross-revision-parity.json').exists(),'production':{},'diagnostic':{},'callback':{}}
for variant in ('base','p6','p7','combined'):
 rs=selected('production',variant,'default_batch')
 if rs:
  result['production'][variant]={'wall':stats([r['wall_seconds'] for r in rs]),'phase_medians':{field:statistics.median(r[field] for r in rs) for field in ('parse_seconds','context_seconds','inventory_seconds','report_seconds')},'peak_rss_mib':stats([r['rss_peak_bytes']/2**20 for r in rs]),'runs':sorted({r['runs'] for r in rs}),'checked_file_counts':sorted({len(r['checked_paths']) for r in rs})}
 for condition in ('batch_graph_off','batch_graph_on'):
  rs=selected('diagnostic',variant,condition)
  if rs:result['diagnostic'][variant+'-'+condition]=stats([r['wall_seconds'] for r in rs])
for condition in ('authoritative_only','plain','noop','probe'):
 rs=selected('callback','combined',condition)
 if rs:
  result['callback'][condition]={'wall':stats([r['wall_seconds'] for r in rs]),'preliminary':stats([r['preliminary_seconds'] for r in rs]),'authoritative':stats([r['authoritative_seconds'] for r in rs]),'peak_rss_mib':stats([r['rss_peak_bytes']/2**20 for r in rs])}
  if condition=='probe':result['callback'][condition]['probe_summaries']=[r['probe'] for r in rs]
base=selected('production','base','default_batch')
for variant in ('p6','p7','combined'):
 rs=selected('production',variant,'default_batch')
 if len(base)==len(rs)==4:
  result['production'][variant]['median_change_percent']=(statistics.median(r['wall_seconds'] for r in rs)/statistics.median(r['wall_seconds'] for r in base)-1)*100
  result['production'][variant]['paired_trial_change_percent']=[(c['wall_seconds']/b['wall_seconds']-1)*100 for b,c in zip(base,rs,strict=True)]
(ROOT/'aggregate-results.json').write_text(json.dumps(result,indent=2),encoding='utf-8')
print(json.dumps({'completed':result['campaign_completed'],'production':{k:v['wall'] for k,v in result['production'].items()},'diagnostic':result['diagnostic'],'callback':{k:v['wall'] for k,v in result['callback'].items()}},indent=2))
