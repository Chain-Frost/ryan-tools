from pathlib import Path
import hashlib,json,os,subprocess,sys,time
ROOT=Path(__file__).absolute().parent
CONFIG=json.loads((ROOT/'config.json').read_text())
ENV=os.environ.copy();ENV.update(TEMP=str(ROOT/'temp'),TMP=str(ROOT/'temp'),PYTHONDONTWRITEBYTECODE='1')
variants=list(CONFIG['sources']);progress=[]

def save():
 (ROOT/'progress.json').write_text(json.dumps(progress,indent=2),encoding='utf-8')

def check_controls():
 old=json.loads((ROOT/'provenance.json').read_text())
 for key in ('batch','tcf'):
  p=Path(CONFIG[key]); current=hashlib.sha256(p.read_bytes()).hexdigest()
  if current!=old[key]['sha256']:raise RuntimeError('Workload control input changed')

def call(stage,variant,trial,condition):
 check_controls()
 label=f'{stage}-{variant}-{trial}-{condition}'
 row={'label':label,'status':'running','started_epoch':time.time()};progress.append(row);save()
 print('START '+label,flush=True)
 with (ROOT/(label+'.log')).open('w',encoding='utf-8') as out:
  process=subprocess.run([sys.executable,str(ROOT/'driver.py'),stage,variant,str(trial),condition],env=ENV,stdout=out,stderr=subprocess.STDOUT)
 row.update(status='passed' if process.returncode==0 else 'failed',exit_code=process.returncode,finished_epoch=time.time());save()
 print('DONE '+label+' exit='+str(process.returncode),flush=True)
 if process.returncode:raise RuntimeError('Campaign stopped at '+label)
 check_controls()

try:
 for variant in variants:call('gate',variant,2,'full')
 # Compare every retained materialised identity and command/reference semantic hash.
 reference=ROOT/'reports'/'gate-base-2-full'
 for variant in variants[1:]:
  candidate=ROOT/'reports'/f'gate-{variant}-2-full'
  for mode in ('eager','deferred','batch_graph_off','batch_graph_on'):
   a=json.loads((reference/(mode+'-identities.json')).read_text());b=json.loads((candidate/(mode+'-identities.json')).read_text())
   if a!=b:raise RuntimeError('Cross-revision complete identities differ: '+variant+'/'+mode)
   a=json.loads((reference/(mode+'-stats.json')).read_text());b=json.loads((candidate/(mode+'-stats.json')).read_text())
   for field in ('command_provenance','reference_provenance'):
    if a[field]!=b[field]:raise RuntimeError('Cross-revision semantic provenance differs: '+variant+'/'+mode+'/'+field)
  if json.loads((reference/'public-preflight.json').read_text())!=json.loads((candidate/'public-preflight.json').read_text()):raise RuntimeError('Cross-revision public preflight differs')
 (ROOT/'cross-revision-parity.json').write_text(json.dumps({'passed':True,'sources':CONFIG['sources'],'all_four_full_modes':True,'public_preflight':True},indent=2))
 # Primary full public production timing: four balanced revision rotations.
 for trial in range(4):
  order=variants[trial:]+variants[:trial]
  for variant in order:call('production',variant,trial,'default_batch')
 # Frozen diagnostic graph policy: alternate mode order and rotate revision order.
 for trial in range(4):
  order=variants[trial:]+variants[:trial]
  modes=['batch_graph_off','batch_graph_on'] if trial%2==0 else ['batch_graph_on','batch_graph_off']
  for variant in order:
   for mode in modes:call('diagnostic',variant,trial,mode)
 # Read-ahead remains a double-collection pilot; all four boundaries include full authoritative completion.
 conditions=['authoritative_only','plain','noop','probe']
 for trial in range(4):
  for condition in conditions[trial:]+conditions[:trial]:call('callback','combined',trial,condition)
 check_controls()
 (ROOT/'completed.json').write_text(json.dumps({'completed':True,'passed':True,'steps':len(progress),'protocol':'protocol.json'},indent=2))
 print('CAMPAIGN COMPLETE',flush=True)
except BaseException as exc:
 (ROOT/'stopped.json').write_text(json.dumps({'completed':False,'passed':False,'error':str(exc),'steps':len(progress)},indent=2))
 raise
