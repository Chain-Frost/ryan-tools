"""Read-only preflight worker; no TUFLOW execution or model mutation."""
import time
entry_counter = time.perf_counter()
import argparse
import dataclasses
import json
import sys
from pathlib import Path
parser = argparse.ArgumentParser(description=__doc__)
parser.add_argument('--config', type=Path, required=True)
parser.add_argument('--name', required=True)
parser.add_argument('--start', type=int, required=True)
parser.add_argument('--stop', type=int, required=True)
args = parser.parse_args()
config = json.loads(args.config.read_text(encoding='utf-8-sig'))
source = Path(config['parser_source']).absolute()
consumer = Path(config['consumer_source']).absolute()
sys.path[:0] = [str(source), str(consumer)]
import psutil
import pytuflow
import ryan_library
from loguru import logger
from ryan_library.orchestrators.tuflow.input_preflight import RunRequest, check_simulation_inputs
# Physical identity is required to verify the actual imported source across links.
assert Path(pytuflow.__file__).resolve().is_relative_to(source.resolve())
assert Path(ryan_library.__file__).resolve().is_relative_to(consumer.resolve())
logger.remove()
all_flags = json.loads(Path(config['runs_json']).read_text())
assert 0 <= args.start < args.stop <= len(all_flags)
runs = []
for index in range(args.start, args.stop):
    tokens = all_flags[index].split()
    runs.append(RunRequest(index, dict(zip((x.lstrip('-') for x in tokens[::2]), tokens[1::2]))))
ready_counter = time.perf_counter()
cpu_start = time.process_time()
report = check_simulation_inputs(Path(config['tcf']), runs, **config['options'])
api_end_counter = time.perf_counter()
report_start = time.perf_counter()
output = Path(config['output']).absolute()
report.write_csv(output / (args.name + '.csv'))
finished_counter = time.perf_counter()
cpu_seconds = time.process_time() - cpu_start
memory = psutil.Process().memory_info()
result = {field.name: getattr(report, field.name) for field in dataclasses.fields(report)
          if isinstance(getattr(report, field.name), (int, float))}
result.update(start=args.start, stop=args.stop, passed=report.passed, issues=len(report.issues),
              paths=len(report.checked_paths), entry_counter=entry_counter, ready_counter=ready_counter,
              api_end_counter=api_end_counter, finished_counter=finished_counter,
              imports_and_setup_seconds=ready_counter-entry_counter,
              api_seconds=api_end_counter-ready_counter, report_seconds=finished_counter-report_start,
              worker_seconds=finished_counter-ready_counter, cpu_seconds=cpu_seconds,
              peak_working_set_bytes=memory.peak_wset)
(output / (args.name + '.json')).write_text(json.dumps(result, indent=2))
(output / (args.name + '.paths.json')).write_text(json.dumps(sorted(report.checked_paths), indent=2))
assert report.passed and report.run_count == args.stop-args.start
assert report.csv_column_selections_checked == report.run_count
assert report.dependency_fallback_runs == 0
print('Read-only worker completed', report.run_count, 'contexts', flush=True)
