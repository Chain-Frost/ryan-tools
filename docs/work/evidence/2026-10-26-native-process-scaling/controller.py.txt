from pathlib import Path
import csv,hashlib,json,os,statistics,subprocess,sys,time
p=Path(os.environ["PRIVATE_PROCESS_BENCHMARK_ROOT"]).absolute();ssd=Path(os.environ["PRIVATE_SSD_BENCHMARK_ROOT"]).absolute();source=Path(os.environ["PRIVATE_SOURCE_SNAPSHOT_ROOT"]).absolute()
source_hashes=json.loads((source/'snapshot-hashes.json').read_text());model_hashes=json.loads((ssd/'source-control-hashes.json').read_text());metadata=json.loads((ssd/'input-metadata-before.json').read_text())
def guard():
 for file,expected in source_hashes.items():assert hashlib.sha256((source/file).read_bytes()).hexdigest()==expected,'Snapshot drift'
 for file,expected in model_hashes.items():assert hashlib.sha256(Path(file).read_bytes()).hexdigest()==expected,'Model source drift'
 for file,expected in metadata.items():
  st=Path(file).stat();assert {'size':st.st_size,'mtime_ns':st.st_mtime_ns}==expected,'Input drift'
# Following links is deliberate: compare actual file identities, including junctions.
expected_paths={str(Path(x).resolve()).casefold() for x in metadata}
def trial(workers,contexts,name):
 guard();launched=[];begin=time.perf_counter()
 try:
  for index in range(workers):
   start=contexts*index//workers;stop=contexts*(index+1)//workers;worker_name=f'{name}-worker-{index}';log=(p/(worker_name+'.log')).open('w');launch=time.perf_counter()
   proc=subprocess.Popen([sys.executable,str(p/'child.py'),'--config',str(p/'config.json'),'--name',worker_name,'--start',str(start),'--stop',str(stop)],stdout=log,stderr=subprocess.STDOUT)
   launched.append((proc,log,worker_name,launch))
  results=[];paths=set();coverage=[];issue_rows=[];header=None
  for proc,log,worker_name,launch in launched:
   code=proc.wait(timeout=180);log.close();assert code==0,'Worker failure: inspect private logs'
   row=json.loads((p/(worker_name+'.json')).read_text());row['launch_offset_seconds']=launch-begin;row['startup_to_ready_seconds']=row['ready_counter']-launch;row['ready_offset_seconds']=row['ready_counter']-begin;results.append(row)
   paths.update(json.loads((p/(worker_name+'.paths.json')).read_text()));coverage.extend(range(row['start'],row['stop']))
   with (p/(worker_name+'.csv')).open(encoding='utf-8-sig',newline='') as f:
    records=list(csv.reader(f));header=records[0];issue_rows.extend(records[1:])
  with (p/(name+'-combined.csv')).open('w',encoding='utf-8-sig',newline='') as f:
   writer=csv.writer(f);writer.writerow(header);writer.writerows(issue_rows)
  elapsed=time.perf_counter()-begin
  assert coverage==list(range(contexts)) and not issue_rows
  if contexts==480:assert {str(Path(x).resolve()).casefold() for x in paths}==expected_paths
  guard()
  result={'name':name,'workers':workers,'contexts':contexts,'end_to_end_seconds':elapsed,'passed':True,'unique_paths':len(paths),'startup_until_all_ready_seconds':max(x['ready_offset_seconds'] for x in results),'total_worker_cpu_seconds':sum(x['cpu_seconds'] for x in results),'summed_worker_peak_working_set_bytes':sum(x['peak_working_set_bytes'] for x in results),'fragment_evaluations':sum(x['dependency_fragment_evaluations'] for x in results),'fragment_reuses':sum(x['dependency_fragment_reuses'] for x in results),'header_reads':sum(x['csv_headers_checked'] for x in results),'column_selections':sum(x['csv_column_selections_checked'] for x in results),'workers_data':results}
  (p/(name+'.json')).write_text(json.dumps(result,indent=2));print(name,round(elapsed,4),'seconds; CPU',round(result['total_worker_cpu_seconds'],4),'ready',round(result['startup_until_all_ready_seconds'],4),flush=True);return result
 finally:
  for proc,log,worker_name,launch in launched:
   if proc.poll() is None:proc.terminate();proc.wait()
   log.close()
assert json.loads((p/'parity.json').read_text())['passed']
calibrations=[trial(1,count,f'initialization-{count}') for count in [1,10,160]]
(p/'initialization.json').write_text(json.dumps(calibrations,indent=2))
results=[]
for block in [1,2]:
 for position,workers in enumerate([1,3,3,1],1):
  result=trial(workers,480,f'block-{block}-{position}-{workers}');result.update(block=block,position=position);results.append(result);(p/'progress.json').write_text(json.dumps(results,indent=2))
print('One-versus-three matched campaign complete',flush=True)
