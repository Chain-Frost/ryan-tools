"""Private-config native measurement; profiles are separate from production timings."""
from __future__ import annotations
import argparse
import cProfile
import dataclasses
import json
import pstats
import sys
import time
from pathlib import Path

parser = argparse.ArgumentParser(description=__doc__)
parser.add_argument('--config', type=Path, required=True)
parser.add_argument('--revision', choices=['A', 'B'], required=True)
parser.add_argument('--name', required=True)
parser.add_argument('--profile', action='store_true')
args = parser.parse_args()
config = json.loads(args.config.read_text(encoding='utf-8-sig'))
output = Path(config['output']).absolute()
parser_root = Path(config['parser_sources'][args.revision]).absolute()
consumer_root = Path(config['consumer_source']).absolute()
sys.path[:0] = [str(parser_root), str(consumer_root)]
import psutil
import pytuflow
import ryan_library
from loguru import logger
from ryan_library.orchestrators.tuflow.input_preflight import RunRequest, check_simulation_inputs
# Follow links deliberately to verify actual imported modules are the frozen snapshots.
assert Path(pytuflow.__file__).resolve().is_relative_to(parser_root.resolve())
assert Path(ryan_library.__file__).resolve().is_relative_to(consumer_root.resolve())
logger.remove()
flags = json.loads(Path(config['runs_json']).read_text())
runs = []
for index, text in enumerate(flags):
    tokens = text.split()
    runs.append(RunRequest(index, dict(zip((x.lstrip('-') for x in tokens[::2]), tokens[1::2]))))
assert len(runs) == config['expected_runs']
profile = cProfile.Profile() if args.profile else None
if profile is not None:
    profile.enable()
wall_start = time.perf_counter()
cpu_start = time.process_time()
report = check_simulation_inputs(Path(config['tcf']), runs, **config['options'])
preflight_seconds = time.perf_counter() - wall_start
preflight_cpu_seconds = time.process_time() - cpu_start
report_start = time.perf_counter()
report.write_csv(output / (args.name + '.csv'))
report_seconds = time.perf_counter() - report_start
end_to_end_seconds = time.perf_counter() - wall_start
cpu_seconds = time.process_time() - cpu_start
if profile is not None:
    profile.disable()
    profile.dump_stats(str(output / (args.name + '.prof')))
    records = []
    for (filename, line, function), (primitive, calls, own, inclusive, _) in pstats.Stats(profile).stats.items():
        file = filename.replace('\\', '/')
        records.append({'module': file.rsplit('/', 1)[-1], 'line': line, 'function': function,
                        'calls': calls, 'primitive_calls': primitive, 'self_seconds': own,
                        'inclusive_seconds': inclusive,
                        'application': '/pytuflow/' in file.lower() or '/ryan_library/' in file.lower()})
    records.sort(key=lambda row: row['self_seconds'], reverse=True)
    (output / (args.name + '.profile.json')).write_text(json.dumps(records, indent=2))
mem = psutil.Process().memory_info()
result = {field.name: getattr(report, field.name) for field in dataclasses.fields(report)
          if isinstance(getattr(report, field.name), (int, float))}
result.update(revision=args.revision, instrumented=args.profile, passed=report.passed,
              issues=len(report.issues), paths=len(report.checked_paths),
              preflight_seconds=preflight_seconds, preflight_cpu_seconds=preflight_cpu_seconds,
              end_to_end_seconds=end_to_end_seconds, cpu_seconds=cpu_seconds,
              cpu_wall_ratio=cpu_seconds / end_to_end_seconds,
              report_seconds=report_seconds, rss_bytes=mem.rss, peak_working_set_bytes=mem.peak_wset)
(output / (args.name + '.json')).write_text(json.dumps(result, indent=2))
# Raw paths are private; output is explicitly outside repository/model directories.
(output / (args.name + '.paths.json')).write_text(json.dumps(sorted(report.checked_paths), indent=2))
assert report.passed and len(report.checked_paths) == config['expected_paths']
assert report.csv_headers_checked == config['expected_headers']
assert report.csv_column_selections_checked == config['expected_selections']
assert report.dependency_fallback_runs == 0
print(json.dumps(result), flush=True)
